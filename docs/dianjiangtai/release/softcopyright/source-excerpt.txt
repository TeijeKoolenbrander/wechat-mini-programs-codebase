function beijingDateKey(nowMs) {

  return Math.floor((nowMs + BJT_OFFSET_MS) / DAY_MS) * DAY_MS - BJT_OFFSET_MS;
}

  return Math.floor((nowMs - beijingMidnightMs(nowMs)) / 1000);

function dateKeyToMs(key) {
  const y = +key.slice(0, 4);
  const m = +key.slice(4, 6);
  const d = +key.slice(6, 8);
  return Date.UTC(y, m - 1, d) - BJT_OFFSET_MS;
}
/** The key `n` days after `key` (n may be negative). */
function addDays(key, n) {
  return beijingDateKey(dateKeyToMs(key) + n * DAY_MS + 12 * 3600e3);
}
/** Whole days from key a to key b. */
function daysBetween(a, b) {
  return Math.round((dateKeyToMs(b) - dateKeyToMs(a)) / DAY_MS);
}
/** Milliseconds until the next 00:00 BJT. */
function msUntilBeijingMidnight(nowMs) {
  return beijingMidnightMs(nowMs) + DAY_MS - nowMs;
}

  return daysBetween(launchKey, dateKey) + 1;

/**
 * Seeded randomness (spec 5.1). Pure JS, no wx, no DOM: this file is copied
 * into the cloud functions, so client and server draw identical streams.
 *
 *   sub(seed, name) = mulberry32(hash32(seed + ":" + name))
 */
/** mulberry32: same seed, same sequence, forever. Returns () => float in [0,1). */
function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
/** FNV-1a over UTF-16 code units, then a murmur3 finaliser. Returns uint32. */
function hash32(str) {
  const s = String(str);
  let h = 0x811c9dc5;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193);
  }
  h ^= h >>> 16;
  h = Math.imul(h, 0x85ebca6b);
  h ^= h >>> 13;
  h = Math.imul(h, 0xc2b2ae35);
  h ^= h >>> 16;
  return h >>> 0;
}
/** A named sub-stream of a run seed. */
function sub(seed, name) {
  return mulberry32(hash32(seed + ':' + name));
}
/** Integer in [0, maxExclusive). */
function int(rng, maxExclusive) {
  return Math.floor(rng() * maxExclusive);
}
/** A uniformly random element of `arr`. */
function pick(rng, arr) {
  return arr[int(rng, arr.length)];
}
/** Fisher-Yates, in place. Returns the same array. */
function shuffle(rng, arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = int(rng, i + 1);
    const t = arr[i];
    arr[i] = arr[j];
    arr[j] = t;

/** Collect every string value in an object tree. */
function collectStrings(x, out) {
  if (typeof x === 'string') out.push(x);
  else if (Array.isArray(x)) x.forEach((y) => collectStrings(y, out));
  else if (x && typeof x === 'object') Object.keys(x).forEach((k) => collectStrings(x[k], out));
  return out;
}

/**

 * `cfg` = {respin: 'none'|'random'|'greedy'|'greedyNoisy', respinP, gainBelow, takeBest}
 */
function playRound(run, model, rng, cfg) {
  draft.apply(run, { round: run.round, type: 'flip' });
  // --- re-spins ---
  if (cfg.respin === 'random') {
    // each available re-spin is used with probability respinP
    for (const dim of ['faction', 'era']) {
      const info = draft.respinInfo(run, dim);
      if (info.enabled && info.nextSource === 'free' && rng() < cfg.respinP) doRespin(run, dim);
    }
  } else if (cfg.respin === 'greedy' || cfg.respin === 'greedyNoisy') {
    for (let guard = 0; guard < 2; guard++) {
      const dims = availableRespins(run);
      if (dims.length === 0) break;
      const opts = options(run);
      const best = opts.length ? opts[0].score : 0;
      if (best >= cfg.gainBelow) break;
      if (cfg.respin === 'greedyNoisy' && rng() >= cfg.respinP) break;
      doRespin(run, dims[dims.length === 1 ? 0 : int(rng, dims.length)]);
    }

----------------------------------------------------------------------------------------------------
  }
  // --- pick ---
  const opts = options(run);
  if (model === 'random') {
    const pool = draft.currentPool(run);
    const g = pool[int(rng, pool.length)];
    const open = [];
    for (let s = 0; s < SLOTS.length; s++) if (!run.slots[SLOTS[s]]) open.push(s);
    doPick(run, g, open[int(rng, open.length)]);

      }

      return;

function playRun(model, seed, contentVersion, mode, rng, scn, opts) {
  const sc = scn || makeScenario(seed, contentVersion, opts && opts.configVersion);
  let m = model;
  if (m === 'sensible') m = rng() < 0.5 ? 'naive' : 'greedy';
  const run = draft.createRun({ seed, contentVersion, mode, configVersion: sc.config.configVersion, 
    rounds: opts && opts.rounds });

  for (let r = 0; r < sc.config.rounds; r++) {
    if (r > 0) draft.nextRound(run);
    playRound(run, m, rng, cfg);
  }

/**
 * Props of `name` reduced to the catalogue: only declared keys (plus the common ones), each type/en
    um checked, everything
 * else dropped (prototype keys included). Returns null for an event name that is not in the catalog
    ue. Never throws.
 */
function cleanProps(name, props) {
  if (typeof name !== 'string' || !has(CATALOGUE, name)) return null;
  const spec = CATALOGUE[name];
  const out = {};
  if (!props || typeof props !== 'object' || Array.isArray(props)) return out;
  const keys = Object.keys(props);
  for (let i = 0; i < keys.length; i++) {
    const k = keys[i];
    const s = has(spec, k) ? spec[k] : has(COMMON, k) ? COMMON[k] : null;
    if (!s) continue;
    const val = checkValue(s, props[k]);
    if (val !== undefined) out[k] = val;
  }
  return out;
}

async function purge(col, cond, deadline) {
  let total = 0;
  while (now() < deadline) {
    const r = await store.db().collection(col).where(cond).remove();
    const n = (r && r.stats && r.stats.removed) || 0;
    total += n;
    if (n === 0) break;

// track: batched analytics (spec §15). <= 50 events per call, rate limited per openid (300 events a
     minute).
// Identity is the context OPENID. Unknown event names are dropped (the batch never fails).
// L2: props are an ALLOWLIST per event: only the keys core/events.js CATALOGUE[name] lists (plus CO
    MMON), each checked for
// its type or enum; every other key is dropped, so a modified client cannot store free text, identi
    fiers or oversized
// values (AC 60: events never contain an openGId or groupRef). The event itself is still stored wit
    hout the rejected keys.
const store = require('./common/store');
const limits = require('./common/limits');
const { fail, now, isPlainObject, isInt } = require('./common/util');
const events = require('./core/events');
const MAX_BATCH = 50;
const MAX_PROPS_BYTES = 1024;
const SESSION_RE = /^[A-Za-z0-9_-]{1,64}$/;
exports.main = store.handler(async (event, { openid }) => {
  const list = event.events;
  if (!Array.isArray(list) || list.length === 0 || list.length > MAX_BATCH) fail('INVALID_ACTIONS', 
    `events must be 1-${MAX_BATCH}`);
  const ts = now();
  const docs = [];
  for (const e of list) {
    if (!isPlainObject(e) || typeof e.name !== 'string') continue; // drop unknown, never fail the b
    atch
    const raw = isPlainObject(e.props) ? e.props : {};
    const props = events.cleanProps(e.name, raw);
    if (!props) continue; // not a client-sendable event (server-only names included)
    // the client sends sessionId inside props; accept the top-level field first, store it top-level
     only
    const rawSession = e.sessionId !== undefined ? e.sessionId : raw.sessionId;
    if (JSON.stringify(props).length > MAX_PROPS_BYTES) continue;
    docs.push({
      openid, sessionId: typeof rawSession === 'string' && SESSION_RE.test(rawSession) ? rawSession 
    : null, name: e.name, props, ts,
      clientTs: isInt(e.clientTs, 0, 4102444800000) ? e.clientTs : null,
    });
  }
  await Promise.all(docs.map((data) => store.db().collection('events').add({ data })));
  return { ok: true, accepted: docs.length };
}, { rate: limits.rateOf('track') });
