/**
 * Beijing date helpers (spec 5.2, AC 66). Explicit UTC+8 arithmetic: never
 * uses local-time getters, so the answer is the same whatever TZ the process
 * (or the device) runs in.
 */
const BJT_OFFSET_MS = 8 * 3600e3;
const DAY_MS = 86400e3;
/** 'YYYYMMDD' of the Beijing date at epoch-ms `nowMs`. */
function beijingDateKey(nowMs) {
  return new Date(nowMs + BJT_OFFSET_MS).toISOString().slice(0, 10).replace(/-/g, '');
}
/** Epoch-ms of 00:00 Beijing time on the date of `nowMs`. */
function beijingMidnightMs(nowMs) {
  return Math.floor((nowMs + BJT_OFFSET_MS) / DAY_MS) * DAY_MS - BJT_OFFSET_MS;
}
/** Seconds after 00:00 BJT (board tie-break `s`, spec 11.4). */
function secondsSinceBeijingMidnight(nowMs) {
  return Math.floor((nowMs - beijingMidnightMs(nowMs)) / 1000);
}
/** Epoch-ms of 00:00 BJT for a 'YYYYMMDD' key. */
function dateKeyToMs(key) {
  const y = +key.slice(0, 4);
  const m = +key.slice(4, 6);
  const d = +key.slice(6, 8);
  return Date.UTC(y, m - 1, d) - BJT_OFFSET_MS;
}
/** The key `n` days after `key` (n may be negative). */

/** FNV-1a over UTF-16 code units, then a murmur3 finaliser. Returns uint32. */
function hash32(str) {
  const s = String(str);
  let h = 0x811c9dc5;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193);
  }
  h ^= h >>> 16;
  h = Math.imul(h, 0x85ebca6b);
  h ^= h >>> 13;
  h = Math.imul(h, 0xc2b2ae35);
  h ^= h >>> 16;
  return h >>> 0;
}
/** A named sub-stream of a run seed. */
function sub(seed, name) {
  return mulberry32(hash32(seed + ':' + name));
}
/** Integer in [0, maxExclusive). */
function int(rng, maxExclusive) {
  return Math.floor(rng() * maxExclusive);
}
/** A uniformly random element of `arr`. */
function pick(rng, arr) {
  return arr[int(rng, arr.length)];
}
/** Fisher-Yates, in place. Returns the same array. */
function shuffle(rng, arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = int(rng, i + 1);
    const t = arr[i];
    arr[i] = arr[j];
    arr[j] = t;
  }
  return arr;
}
/** A fresh random uint32 from Math.random (client-side free-play seeds only). */
function randomSeed() {
  return Math.floor(Math.random() * 4294967296) >>> 0;
}

function emptyHist() {
  return new Array(BINS).fill(0);

function histTotal(hist) {
  let n = 0;
  for (let i = 0; i < hist.length; i++) n += hist[i] || 0;
  return n;
}
/** Median of a sorted-ascending read of a histogram (spec 3.2). null when empty. O(101). */
function histMedian(hist) {
  const M = histTotal(hist);
  if (M === 0) return null;
  const lo = Math.floor((M - 1) / 2);
  const hi = Math.floor(M / 2);
  let seen = 0;
  let a = null;
  let b = null;
  for (let v = 0; v < hist.length; v++) {
    const c = hist[v] || 0;
    if (c === 0) continue;
    if (a === null && lo < seen + c) a = v;
    if (b === null && hi < seen + c) b = v;
    seen += c;
    if (a !== null && b !== null) break;
  }
  return (a + b) / 2;
}

/** Mean absolute deviation of `values` from `center`. 0 for an empty list. */
function madOf(values, center) {
  if (values.length === 0) return 0;
  let s = 0;
  for (let i = 0; i < values.length; i++) s += Math.abs(values[i] - center);
  return s / values.length;
}
/** MAD of a histogram's real positions about `center`. 0 when empty. */
function histMad(hist, center) {
  const M = histTotal(hist);
  if (M === 0) return 0;
  let s = 0;
  for (let v = 0; v < hist.length; v++) if (hist[v]) s += hist[v] * Math.abs(v - center);
  return s / M;
}
/**
 * National reference R_c (spec 3.3, PD-4): median of the real positions plus `priorVotes` virtual v
    otes at the
 * editor anchor. With no real positions it is the anchor. Does not mutate `hist`.
 */
function nationRef(hist, anchor, priorVotes) {
  const W = priorVotes === undefined ? CONFIG.score.priorVotes : priorVotes;
  const h = hist.slice();
  const a = Math.round(clampPos(anchor));
  h[a] = (h[a] || 0) + W;
  return histMedian(h);
}
/** Share of real positions within +-agreeRadius of p (integer percent), or null below agreeMinN (sp
    ec 3.3). */
function agreePct(hist, p) {
  const M = histTotal(hist);
  if (M < CONFIG.score.agreeMinN) return null;
  const r = CONFIG.score.agreeRadius;
  let c = 0;
  for (let v = Math.max(0, p - r); v <= Math.min(100, p + r); v++) c += hist[v] || 0;
  return Math.round((100 * c) / M);
}
/** 21 display bins of the national distribution (spec 6.4). Centred: bin i holds positions 5i-2..5i
    +2, so both ends are half-width. */
function bin21(pos) {
  return Math.floor((pos + 2) / 5);
}
function distribution21(hist) {
  const out = new Array(21).fill(0);
  for (let v = 0; v < hist.length; v++) if (hist[v]) out[bin21(v)] += hist[v];
  return out;
}

 * Card index with the widest national spread (spec 3.3): highest MAD of real positions about R_c; t
    ies go to the
 * lower index; null when no card has an answer.
 * @param {number[][]} hists 5 x 101   @param {number[]} refs 5 reference values

function mostDivisive(hists, refs) {
  let best = null;
  let bestMad = -1;
  for (let c = 0; c < hists.length; c++) {
    if (histTotal(hists[c]) === 0) continue;
    const m = histMad(hists[c], refs[c]);
    if (m > bestMad) {
      bestMad = m;
      best = c;
    }
  }

/** openapi results carry errCode (0 = ok); a non-zero code is a failure even when the SDK did not t
    hrow */
function okResult(res, what) {
  if (!res) throw new Error(what + ': empty response');
  // the cloud SDK answers errCode / errMsg, the HTTPS API errcode / errmsg: read both
  const code = res.errCode !== undefined && res.errCode !== null ? res.errCode : res.errcode;
  if (code !== undefined && code !== null && code !== 0) {
    throw Object.assign(new Error(what + ': errCode ' + code + ' ' + (res.errMsg || res.errmsg || ''
    )), { errCode: code });
  }
  return res;
}
